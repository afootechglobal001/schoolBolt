CREATE OR REPLACE VIEW BRANCH_VIEW AS
SELECT 
a.*, b.stateName, c.lgaName, CONCAT(d.firstName, ' ', d.lastName) AS managerName, e.statusName, f.termName, g.schoolCategoryName
FROM 
BRANCHES_TAB a
JOIN SETUP_STATES_TAB b ON a.stateId=b.stateId
JOIN SETUP_STATE_LGA_TAB c ON a.lgaId=c.lgaId
JOIN STAFF_TAB d ON a.clientId=d.clientId AND a.managerId=d.staffId
JOIN SETUP_STATUS_TAB e ON a.statusId=e.statusId
JOIN SETUP_TERM_TAB f ON a.termId=f.termId
JOIN SETUP_SCHOOL_CATEGORY_TAB g ON a.schoolCategoryId=g.schoolCategoryId;

CREATE OR REPLACE VIEW PARENT_VIEW AS
SELECT a.*, b.name, c.statusName 
FROM 
PARENTS_TAB a
JOIN BRANCHES_TAB b ON a.clientId=b.clientId AND a.branchId=b.branchId
JOIN SETUP_STATUS_TAB c ON a.statusId=c.statusId;


CREATE OR REPLACE VIEW STAFF_VIEW AS
SELECT 
a.*, b.titleName, c.genderName, d.stateName, e.lgaName, f.name AS branchName, g.roleName, g.rolePermissionIds, h.statusName
FROM 
STAFF_TAB a
JOIN SETUP_TITLE_TAB b ON a.titleId=b.titleId
JOIN SETUP_GENDER_TAB c ON a.genderId=c.genderId
JOIN SETUP_STATES_TAB d ON a.stateId=d.stateId
JOIN SETUP_STATE_LGA_TAB e ON a.lgaId=e.lgaId
JOIN BRANCHES_TAB f ON a.clientId=f.clientId AND a.branchId=f.branchId
JOIN ROLE_TAB g ON a.clientId=g.clientId AND a.roleId=g.roleId
JOIN SETUP_STATUS_TAB h ON a.statusId=h.statusId;

CREATE OR REPLACE VIEW STUDENT_VIEW AS
SELECT a.*, b.name, c.genderName, d.maritalStatusName, e.countryName, f.stateName, g.lgaName, h.statusName, i.officialStudentId 
FROM 
STUDENTS_TAB a
JOIN BRANCHES_TAB b ON a.clientId=b.clientId AND a.branchId=b.branchId
JOIN SETUP_GENDER_TAB c ON a.genderId=c.genderId
JOIN SETUP_MARITAL_STATUS_TAB d ON a.maritalStatusId=d.maritalStatusId
JOIN SETUP_COUNTRY_TAB e ON a.countryId=e.countryId
JOIN SETUP_STATES_TAB f ON a.stateId=f.stateId
JOIN SETUP_STATE_LGA_TAB g ON a.stateId=g.stateId AND a.lgaId=g.lgaId
JOIN SETUP_STATUS_TAB h ON a.statusId=h.statusId
JOIN STUDENTS_CLASS_TAB i ON a.studentId=i.studentId
